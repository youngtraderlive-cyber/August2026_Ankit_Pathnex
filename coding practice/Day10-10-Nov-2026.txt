- name: Setup Docker Container for multiple app
  hosts: all
  become: yes
    tasks:
        - name: Pull Web App Image
           docker_image:
               name: ankit-web-app
               source: pull
        - name: Pull Database App image
           docker_image:
              name: ankit-database-app
              source: pull
        - name: Run Web App Container
           docker_container:
              name: ankit-app
              image: ankit-web-app
              state: started
              published_ports:
                - "8080:80"
        - name: Run Database App Container
          docker_container:
            name: database-app
            image: ankit-database-app
            state: started
            published_ports:
               - "3306:3306"

Jenkins - Build Deploy to Kubernetes

pipeline {
  agent any    
  environment {
      IMAGE_NAME = 'ankit-web-app'
}
   stages {
     stage('Build Docker Image') {
        step {
           script {
             docker.build("${env.IMAGE_|NAME}")
          } 
       }
    }
     stage('Push Docker Image') {
       step {
          script {
            docker.withRegistry('http://docker.io'. 'docker credentials') {
             docker.image("${env.IMAGE_NAME}").push('latest')
             }
          }
       }
    }
     stage('Deploy to kubernetes') {
        step {
          script {
             sh 'kubetl apply -f deployment.yaml'
             }
           }
         }
       }
    }
