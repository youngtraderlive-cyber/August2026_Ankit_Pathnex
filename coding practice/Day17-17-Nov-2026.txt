      stages {
         stage('Build') {
          steps {

           }
        }
          stage('Test') {
           steps {
             echo 'Running test'
            }
         }
           stage('Deploy') {
             steps {
               script {
               if (env.CLUSTER == 'us-east') {
                     sh 'kubectl --context=us-east apply - kubernetes/deployment.yaml'
                    } else {
                       sh 'kubectl --context=us-west apply -f kubernets/deployment.yaml'
                          }
                       }
                   }
               }
           }
       }
