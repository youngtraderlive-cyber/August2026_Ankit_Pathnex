Ansible install helm on server

- name: Install Helm
  hosts: all
  become: yes
    tasks:
      - name: Download Helm
         get_url:
           url: https/get.helm.sh/helm-v3.7.0-Linux-amd64.tar.gz
           dest: /tmp/helm.tar.gz
      - name: Extract Helm
        unarchive:
           src: /tmp/helm.tar.gz
           dest: /usr/local/bin/
           remote_src: yes
      - name: set helm binary permissions
        file:
          path: /usr/local/bin/helm
          mode: '0755'

jenkins- multistage pipeline
pipeline {
  agent any 
     envionment {
       IMAGE_NAME = "ankit-web-app"
    }
       stages {
          stage('Checkout') {
            steps {
              git 'https://github.com/ankit/sample-repo.git'
          }
      }
        stage('Build Docker Image') {
          steps {
              script {
                  docker.build("${env.IMAGE_NAME}")
                 }
              }    
           }
        stage("Push Docker Image") {
           steps {
             script {
                  docker.withRegistry('https://docker.io','docker-credentails') {
                    docker.image("${env.IMAGE_NAME}").push('latest')
                   }
                }
             }
          }
        stage("Deploy to Kubernetes") {
            steps {
              script {
                     sh 'kubectl apply -f deployment.yaml'
                   }
                }
            }
        }
     }


Gitlab CI/CD - Build, Push and Deploy with Helm
stage: 
  - build
  - push
  - deploy

Build:
  stage: build
   script:
        - docker build -t ankit-web-app
Push:
  stage: push
  script:
       - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
       - docker push ankit-web-app
Deploy:
  stage: deploy
  script:
       - helm repo add stable https://chart.helm.sh/stable
       - helm install ankit-nginx stable/nginx-ingress
