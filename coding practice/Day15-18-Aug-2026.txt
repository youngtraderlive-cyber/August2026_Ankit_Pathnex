name: Deploy web application to multiple hosts
  hosts: all
  become: yes
  tasks:
    - name: Copy web app files
      copy:
        src: /path/to/web/app/
        dest: /var/www/html/
    - name: Start web app service
      service:
        name: apache2
        state: started
        enabled: yes
