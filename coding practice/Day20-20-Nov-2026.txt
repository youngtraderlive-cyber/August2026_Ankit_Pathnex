Ansible - Configure firewell with iptable
- name: Configure Firewell with iptable
  hosts: all
  become: yes
     tasks:
       - name: set default policies to drop all traffic
         command: iptable -P INPUT DROP
       - name: Allow SSH traffic
         command: iptables -A tcp --dport 22 -j ACCEPT
       - name: Allow HTTP traffic
         command: iptables -A INPUT -p tcp --doprt 80 -j ACCEPT
       - name: Save iptables configuration
         command: service iptables save

jenkins - secure credential management
pipeline {
  agnet any
  environment {
     REGISTRY_PASSWORD = credentials('docker-registry-password') 
}
   stages {
    stage('Build') {
      steps {
          echo 'Building project...'
          docker.build('ankit-web-app')
       }
    }
     stage('push') {
       steps {
           script {
               docker.withRegistry('https://docker.io','docker-credentials') {
                 docker.image('ankit-web-app').push('latest')
                 }
              }
           }
        }
     }
  }
