- name: Setup nginx with custom config
  hosts: all
  become: yes
      tasks: 
        - name: Insrall Nginx
            yum:
               name: nginx
               state: present
        - name: copy custom nginx.conf
            copy:
               src: nginx.conf
               dest: /etc/nginx/nginx.conf
        - name: Start nginx service
           service: 
               name: nginx
               state: started
               enabled: yes
